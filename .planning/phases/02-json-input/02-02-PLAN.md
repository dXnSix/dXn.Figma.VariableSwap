   ```typescript
   async function runSwap(mappingEntries: MappingEntry[]) {
     // Convert mapping entries to lookup object for efficient access
     // Handles both string and array formats for "old" field
     const colorMapping: { [key: string]: string } = {};
     for (const entry of mappingEntries) {
       if (typeof entry.old === 'string') {
         colorMapping[entry.old] = entry.new;
       } else {
         // Array format: each old value maps to the same new value
         for (const oldValue of entry.old) {
           colorMapping[oldValue] = entry.new;
         }
       }
     }

     const selection = figma.currentPage.selection;
     let changesMade = 0;

     if (selection.length === 0) {
       figma.notify("No nodes selected");
       return;
     }

     for (const node of selection) {
       changesMade = await processNode(node, colorMapping, changesMade);
     }

     figma.notify(`${changesMade} changes made`);
   }
   ```
